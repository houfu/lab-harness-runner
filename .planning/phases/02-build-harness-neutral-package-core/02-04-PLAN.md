from lab_harness_runner import read_task, build_result_dir, write_metrics, score_run
from lab_harness_runner.adapter import TaskSpec, RunResult

read_task(lab_path: Path, task_id: str, run_id: str) -> TaskSpec
  Raises: FileNotFoundError if task.json missing
  Raises: ValueError if task_id is absolute or contains ".."
  Raises: KeyError if "instructions" missing from task.json
  Deliverables source: criteria[].deliverables (NOT top-level "deliverables")

build_result_dir(lab_path: Path, run_id: str) -> tuple[Path, Path]
  Returns: (run_dir, output_dir) where output_dir = run_dir / "output"
  Creates: output_dir.mkdir(parents=True, exist_ok=True)

write_metrics(run_dir: Path, result: RunResult) -> Path
  Returns: run_dir / "metrics.json"
  Safe defaults: int fields → 0, list fields → []

score_run(lab_path: Path, run_id: str, task_id: str, expected_deliverables: list[str], judge_model: str = "claude-sonnet-4-6") -> Path
  Returns: lab_path / "results" / run_id / "scores.json"
  Raises: FileNotFoundError if any deliverable missing from output_dir (BEFORE subprocess call)
